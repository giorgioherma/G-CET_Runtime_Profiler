cmake_minimum_required(VERSION 3.21)
project(GCETStaticLocations LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(gcet_static_core STATIC src/StaticIndex.cpp src/TransitionJournal.cpp)
target_include_directories(gcet_static_core PUBLIC include)
if(MSVC)
  target_compile_options(gcet_static_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(gcet_static_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(gcet_static_core_test tests/StaticIndexTests.cpp)
target_link_libraries(gcet_static_core_test PRIVATE gcet_static_core)
add_test(NAME native_static_index COMMAND gcet_static_core_test)
option(GCET_BUILD_PLUGIN "Build RED4ext plugin (Windows MSVC with SDK)" ${WIN32})
if(GCET_BUILD_PLUGIN)
  if(NOT WIN32)
    message(FATAL_ERROR "The RED4ext plugin targets Windows x64 only")
  endif()
  include(FetchContent)
  set(RED4EXT_HEADER_ONLY OFF CACHE BOOL "" FORCE)
  set(RED4EXT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(red4ext_sdk
    GIT_REPOSITORY https://github.com/WopsS/RED4ext.SDK.git
    GIT_TAG f9ad5334362a8a0815baf58f47615697485feb52
    GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(red4ext_sdk)
  add_library(GCETStaticLocations SHARED src/Plugin.cpp)
  target_link_libraries(GCETStaticLocations PRIVATE gcet_static_core RED4ext::SDK)
  target_include_directories(GCETStaticLocations PRIVATE include)
  set_target_properties(GCETStaticLocations PROPERTIES OUTPUT_NAME "GCETStaticLocations")
endif()
