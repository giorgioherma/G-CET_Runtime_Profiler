cmake_minimum_required(VERSION 3.16)
project(GCETNativeActionGate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(gcet_native_action_gate SHARED src/gcet_action_gate.cpp)
target_include_directories(gcet_native_action_gate PUBLIC include)
if(WIN32)
  set_target_properties(gcet_native_action_gate PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(gcet_native_gate_tests tests/test_action_gate.cpp)
  target_link_libraries(gcet_native_gate_tests PRIVATE gcet_native_action_gate Threads::Threads)
  add_test(NAME gcet_native_gate_tests COMMAND gcet_native_gate_tests)
endif()
